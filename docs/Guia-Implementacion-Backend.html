1.3 Adaptador de Infraestructura para Brevo API
src/common/infrastructure/adapters/out/mail/brevo-mail.adapter.ts

Consume directamente el endpoint REST v3 de Brevo (https://api.brevo.com/v3/smtp/email) usando fetch nativo de Node 20+, con total resiliencia a fallos.

TypeScript — brevo-mail.adapter.ts
import { Injectable, Logger } from '@nestjs/common';
import { MailPort, SendEmailOptions, EmailRecipient } from '../../../../ports/out/mail.port';

@Injectable()
export class BrevoMailAdapter implements MailPort {
  private readonly logger = new Logger(BrevoMailAdapter.name);
  private readonly apiUrl = 'https://api.brevo.com/v3/smtp/email';

  async sendMail(options: SendEmailOptions): Promise {
    const apiKey = process.env.BREVO_API_KEY;
    const senderEmail = process.env.BREVO_SENDER_EMAIL || 'notificaciones@tandea.app';
    const senderName = process.env.BREVO_SENDER_NAME || 'Tandea';

    if (!apiKey) {
      this.logger.warn('BREVO_API_KEY no configurada. Omitiendo envío de correo real.');
      return false;
    }

    // Normalizar lista de destinatarios
    const recipients: { email: string; name?: string }[] = Array.isArray(options.to)
      ? options.to.map((r) => (typeof r === 'string' ? { email: r } : r))
      : [{ email: options.to }];

    const payload = {
      sender: { name: senderName, email: senderEmail },
      to: recipients,
      subject: options.subject,
      htmlContent: options.htmlContent,
      textContent: options.textContent,
    };

    try {
      const response = await fetch(this.apiUrl, {
        method: 'POST',
        headers: {
          'accept': 'application/json',
          'api-key': apiKey,
          'content-type': 'application/json',
        },
        body: JSON.stringify(payload),
      });

      if (!response.ok) {
        const errorBody = await response.text();
        this.logger.error(`Fallo en Brevo API (Status ${response.status}): ${errorBody}`);
        return false;
      }

      this.logger.log(`Correo enviado exitosamente a: ${recipients.map((r) => r.email).join(', ')}`);
      return true;
    } catch (error: any) {
      // Regla de oro del proyecto: el fallo de correo NUNCA debe romper la API
      this.logger.error(`Error de red al conectar con Brevo API: ${error?.message || error}`);
      return false;
    }
  }
}
¿Por qué es superior esta implementación?
  • Cero librerías pesadas adicionales: Usa fetch estándar de Node.js 20+.
  • Resiliencia garantizada: Atrapa errores en un try/catch y retorna false sin arrojar una excepción no controlada que tumbe el registro del usuario.
  • Variables ya configuradas: Lee directamente tus variables BREVO_API_KEY, BREVO_SENDER_EMAIL y BREVO_SENDER_NAME de tu .env.

export type TandaStatus = 'PENDIENTE' | 'ACTIVA' | 'FINALIZADA' | 'SUSPENDIDA' | 'CANCELADA';

export interface TandaProps {
  idTanda?: number;
  name: string;
  description?: string | null;
  startDate: string; // Formato YYYY-MM-DD
  endDate?: string | null;
  contributionAmount: number;
  numParticipants: number;
  numRounds: number;
  graceDays: number;
  hasDeadNumber: boolean;
  idTandaType: number;
  idAdmin: number;
  status?: TandaStatus;
  active?: boolean;
  inviteCode: string;
  createdAt?: Date;
  updatedAt?: Date;
}

export class Tanda {
  constructor(private readonly props: TandaProps) {
    this.validate();
  }

  private validate(): void {
    if (this.props.numParticipants < 2) {
      throw new Error('Una tanda debe tener al menos 2 participantes');
    }
    if (this.props.contributionAmount <= 0) {
      throw new Error('El monto de contribución debe ser mayor a 0');
    }
    if (this.props.graceDays < 0) {
      throw new Error('Los días de gracia no pueden ser negativos');
    }
    if (this.props.numRounds < 1) {
      throw new Error('Debe haber al menos 1 ronda');
    }
  }

  get idTanda(): number | undefined { return this.props.idTanda; }
  get name(): string { return this.props.name; }
  get description(): string | null | undefined { return this.props.description; }
  get startDate(): string { return this.props.startDate; }
  get endDate(): string | null | undefined { return this.props.endDate; }
  get contributionAmount(): number { return this.props.contributionAmount; }
  get numParticipants(): number { return this.props.numParticipants; }
  get numRounds(): number { return this.props.numRounds; }
  get graceDays(): number { return this.props.graceDays; }
  get hasDeadNumber(): boolean { return this.props.hasDeadNumber; }
  get idTandaType(): number { return this.props.idTandaType; }
  get idAdmin(): number { return this.props.idAdmin; }
  get status(): TandaStatus { return this.props.status ?? 'PENDIENTE'; }
  get active(): boolean { return this.props.active ?? true; }
  get inviteCode(): string { return this.props.inviteCode; }

  toPrimitives(): TandaProps {
    return { ...this.props, status: this.status, active: this.active };
  }
}
2.2 Generador de Código de Invitación (Reutilizable)
src/tandas/domain/utils/invite-code.generator.ts

Genera un código legible y sin caracteres ambiguos (como O, 0, I, 1) en formato TND-XXXXXX.

TypeScript — invite-code.generator.ts
import * as crypto from 'crypto';

export class InviteCodeGenerator {
  // Caracteres seguros y legibles sin confusión visual
  private static readonly CHARACTERS = '23456789ABCDEFGHJKLMNPQRSTUVWXYZ';

  /**
   * Genera un código legible con prefijo TND (ej. TND-7K9A2B)
   */
  static generate(length: number = 6): string {
    const bytes = crypto.randomBytes(length);
    let result = '';
    for (let i = 0; i < length; i++) {
      result += this.CHARACTERS[bytes[i] % this.CHARACTERS.length];
    }
    return `TND-${result}`;
  }
}

TypeScript — tanda.service.ts
import { Injectable, Inject, ConflictException, NotFoundException, BadRequestException } from '@nestjs/common';
import { CreateTandaUseCasePort, CreateTandaCommand } from '../../domain/ports/in/create-tanda-use-case.port';
import { TANDA_REPOSITORY_PORT, type TandaRepositoryPort } from '../../domain/ports/out/tanda-repository.port';
import { NUMBER_REPOSITORY_PORT, type NumberRepositoryPort } from '../../domain/ports/out/number-repository.port';
import { Tanda } from '../../domain/entities/tanda.entity';
import { InviteCodeGenerator } from '../../domain/utils/invite-code.generator';

@Injectable()
export class TandaService implements CreateTandaUseCasePort {
  constructor(
    @Inject(TANDA_REPOSITORY_PORT)
    private readonly tandaRepository: TandaRepositoryPort,
    @Inject(NUMBER_REPOSITORY_PORT)
    private readonly numberRepository: NumberRepositoryPort,
  ) {}

  async crearTanda(command: CreateTandaCommand): Promise {
    // 1. Generar código único y verificar colisión en BD
    let inviteCode = '';
    let isUnique = false;
    let attempts = 0;

    while (!isUnique && attempts < 5) {
      inviteCode = InviteCodeGenerator.generate();
      const existing = await this.tandaRepository.findByInviteCode(inviteCode);
      if (!existing) {
        isUnique = true;
      }
      attempts++;
    }

    if (!isUnique) {
      throw new ConflictException('No se pudo generar un código único de tanda. Reintente.');
    }

    // 2. Instanciar y persistir entidad
    const tanda = new Tanda({
      name: command.name,
      description: command.description,
      startDate: command.startDate,
      contributionAmount: command.contributionAmount,
      numParticipants: command.numParticipants,
      numRounds: command.numRounds ?? 1,
      graceDays: command.graceDays ?? 0,
      hasDeadNumber: command.hasDeadNumber ?? false,
      idTandaType: command.idTandaType ?? 1,
      idAdmin: command.idAdmin,
      inviteCode,
      status: 'PENDIENTE',
    });

    const tandaCreada = await this.tandaRepository.create(tanda);

    // 3. Si tiene número muerto configurado, reservar el turno en la vuelta 1
    if (tandaCreada.hasDeadNumber && tandaCreada.idTanda) {
      await this.numberRepository.createNumber({
        idTanda: tandaCreada.idTanda,
        roundNumber: 1,
        numberValue: tandaCreada.numParticipants, // Último turno para el número muerto
        idUser: null,
        isDeadNumber: true,
        status: 'PENDIENTE',
      });
    }

    return tandaCreada;
  }
}

3.2 Lógica de Selección Manual vs Sorteo/Automático
src/tandas/application/services/tanda.service.ts (método asignarNumero)

Algoritmo que valida que el número no esté ocupado o encuentra el primer turno libre.

TypeScript — Algoritmo de asignación
/**
 * Asigna un turno a un usuario en una tanda.
 * Si selectedNumber viene definido -> Modalidad Manual (valida colisión).
 * Si selectedNumber es null/undefined -> Modalidad Automática (primer número libre).
 */
async asignarNumero(
  idTanda: number,
  idUser: number,
  roundNumber: number = 1,
  selectedNumber?: number,
): Promise {
  const tanda = await this.tandaRepository.findById(idTanda);
  if (!tanda) throw new NotFoundException('Tanda no encontrada');

  const occupied = await this.numberRepository.findOccupiedNumbers(idTanda, roundNumber);

  let chosenNumber: number;

  if (selectedNumber !== undefined && selectedNumber !== null) {
    // MODALIDAD MANUAL
    if (selectedNumber < 1 || selectedNumber > tanda.numParticipants) {
      throw new BadRequestException(
        `El número debe estar entre 1 y ${tanda.numParticipants}`
      );
    }
    if (occupied.includes(selectedNumber)) {
      throw new ConflictException(
        `El número ${selectedNumber} ya está tomado por otro participante`
      );
    }
    chosenNumber = selectedNumber;
  } else {
    // MODALIDAD AUTOMÁTICA
    const allNumbers = Array.from({ length: tanda.numParticipants }, (_, i) => i + 1);
    const available = allNumbers.filter((n) => !occupied.includes(n));

    if (available.length === 0) {
      throw new ConflictException('No hay números disponibles en esta tanda');
    }
    chosenNumber = available[0]; // Primer número disponible
  }

  return await this.numberRepository.createNumber({
    idTanda,
    roundNumber,
    numberValue: chosenNumber,
    idUser,
    isDeadNumber: false,
    status: 'PENDIENTE',
  });
}

  }

  // 2. Solo se pueden unir en estado PENDIENTE
  if (tanda.status !== 'PENDIENTE') {
    throw new BadRequestException('La tanda ya inició o no admite nuevos participantes');
  }

  // 3. Validar que el participante no esté ya unido
  const existingRecords = await this.numberRepository.findByUserInTanda(tanda.idTanda!, idUser);
  if (existingRecords.length > 0) {
    throw new ConflictException('Ya estás registrado como participante de esta tanda');
  }

  // 4. Validar cupo
  const currentCount = await this.numberRepository.countParticipants(tanda.idTanda!);
  if (currentCount >= tanda.numParticipants) {
    throw new BadRequestException('La tanda ha alcanzado el cupo máximo de participantes');
  }

  // 5. Asignar el turno
  const assigned = await this.asignarNumero(tanda.idTanda!, idUser, 1, selectedNumber);

  return {
    mensaje: 'Te has unido exitosamente a la tanda',
    tanda: {
      idTanda: tanda.idTanda,
      name: tanda.name,
      contributionAmount: tanda.contributionAmount,
      startDate: tanda.startDate,
    },
    turnoAsignado: assigned.numberValue,
  };
}

TypeScript — drizzle-tanda.repository.ts
import { Injectable, Inject } from '@nestjs/common';
import { NodePgDatabase } from 'drizzle-orm/node-postgres';
import { eq, and, count } from 'drizzle-orm';
import { DATABASE_TOKEN } from '../../../../common/tokens';
import { tandaTable, numberTable } from '../../../../database/tandea.schema';
import { Tanda } from '../../../domain/entities/tanda.entity';
import { TandaRepositoryPort } from '../../../domain/ports/out/tanda-repository.port';
import { NumberRepositoryPort, NumberRecord } from '../../../domain/ports/out/number-repository.port';

@Injectable()
export class DrizzleTandaRepository implements TandaRepositoryPort, NumberRepositoryPort {
  constructor(
    @Inject(DATABASE_TOKEN)
    private readonly db: NodePgDatabase,
  ) {}

  // --- TANDA REPOSITORY ---
  async create(tanda: Tanda): Promise {
    const [row] = await this.db.insert(tandaTable).values({
      name: tanda.name,
      description: tanda.description,
      startDate: tanda.startDate,
      contributionAmount: tanda.contributionAmount.toString(),
      numParticipants: tanda.numParticipants,
      numRounds: tanda.numRounds,
      graceDays: tanda.graceDays,
      hasDeadNumber: tanda.hasDeadNumber,
      idTandaType: tanda.idTandaType,
      idAdmin: tanda.idAdmin,
      status: tanda.status,
      inviteCode: tanda.inviteCode,
    }).returning();

    return new Tanda({
      idTanda: row.idTanda,
      name: row.name,
      description: row.description,
      startDate: row.startDate,
      contributionAmount: Number(row.contributionAmount),
      numParticipants: row.numParticipants,
      numRounds: row.numRounds,
      graceDays: row.graceDays,
      hasDeadNumber: row.hasDeadNumber,
      idTandaType: row.idTandaType,
      idAdmin: row.idAdmin,
      status: row.status as any,
      active: row.active,
      inviteCode: row.inviteCode,
    });
  }

  async findByInviteCode(inviteCode: string): Promise {
    const [row] = await this.db.select().from(tandaTable).where(eq(tandaTable.inviteCode, inviteCode)).limit(1);
    if (!row) return null;
    return new Tanda({
      idTanda: row.idTanda,
      name: row.name,
      description: row.description,
      startDate: row.startDate,
      contributionAmount: Number(row.contributionAmount),
      numParticipants: row.numParticipants,
      numRounds: row.numRounds,
      graceDays: row.graceDays,
      hasDeadNumber: row.hasDeadNumber,
      idTandaType: row.idTandaType,
      idAdmin: row.idAdmin,
      status: row.status as any,
      active: row.active,
      inviteCode: row.inviteCode,
    });
  }

  async findById(idTanda: number): Promise {
    const [row] = await this.db.select().from(tandaTable).where(eq(tandaTable.idTanda, idTanda)).limit(1);
    if (!row) return null;
    return new Tanda({
      idTanda: row.idTanda,
      name: row.name,
      description: row.description,
      startDate: row.startDate,
      contributionAmount: Number(row.contributionAmount),
      numParticipants: row.numParticipants,
      numRounds: row.numRounds,
      graceDays: row.graceDays,
      hasDeadNumber: row.hasDeadNumber,
      idTandaType: row.idTandaType,
      idAdmin: row.idAdmin,
      status: row.status as any,
      active: row.active,
      inviteCode: row.inviteCode,
    });
  }

  async updateStatus(idTanda: number, status: any): Promise {
    await this.db.update(tandaTable).set({ status }).where(eq(tandaTable.idTanda, idTanda));
  }

  // --- NUMBER REPOSITORY ---
  async createNumber(data: NumberRecord): Promise {
    const [row] = await this.db.insert(numberTable).values({
      idTanda: data.idTanda,
      roundNumber: data.roundNumber,
      numberValue: data.numberValue,
      idUser: data.idUser,
      isDeadNumber: data.isDeadNumber,
      status: data.status,
    }).returning();
    return {
      idNumber: row.idNumber,
      idTanda: row.idTanda,
      roundNumber: row.roundNumber,
      numberValue: row.numberValue,
      idUser: row.idUser,
      isDeadNumber: row.isDeadNumber,
      status: row.status as any,
    };
  }

  async findOccupiedNumbers(idTanda: number, roundNumber: number): Promise {
    const rows = await this.db
      .select({ numberValue: numberTable.numberValue })
      .from(numberTable)
      .where(and(eq(numberTable.idTanda, idTanda), eq(numberTable.roundNumber, roundNumber)));
    return rows.map((r) => r.numberValue);
  }

  async findByUserInTanda(idTanda: number, idUser: number): Promise {
    const rows = await this.db
      .select()
      .from(numberTable)
      .where(and(eq(numberTable.idTanda, idTanda), eq(numberTable.idUser, idUser)));
    return rows as any;
  }

  async countParticipants(idTanda: number): Promise {
    const [res] = await this.db
      .select({ total: count() })
      .from(numberTable)
      .where(eq(numberTable.idTanda, idTanda));
    return Number(res?.total || 0);
  }
}
4.4 Controlador HTTP: TandaController

Expone los endpoints protegidos con JwtAuthGuard para crear tanda y unirse.

TypeScript — tanda.controller.ts
import { Controller, Post, Body, UseGuards, Request, HttpCode, HttpStatus } from '@nestjs/common';
import { AuthGuard } from '@nestjs/passport';
import { TandaService } from '../../../application/services/tanda.service';
import { zCreateTanda } from '../../../application/dtos/create-tanda.dto';
import { zJoinTanda } from '../../../application/dtos/join-tanda.dto';

@Controller('tandas')
@UseGuards(AuthGuard('jwt'))
export class TandaController {
  constructor(private readonly tandaService: TandaService) {}

  @Post()
  @HttpCode(HttpStatus.CREATED)
  async crearTanda(@Body() body: unknown, @Request() req: any) {
    const dto = zCreateTanda.parse(body);
    const userId = req.user.idUser || req.user.sub;

    const tanda = await this.tandaService.crearTanda({
      ...dto,
      idAdmin: userId,
    });

    return {
      statusCode: HttpStatus.CREATED,
      mensaje: 'Tanda creada exitosamente',
      data: tanda.toPrimitives(),
    };
  }

  @Post('unirse')
  @HttpCode(HttpStatus.OK)
  async unirse(@Body() body: unknown, @Request() req: any) {
    const dto = zJoinTanda.parse(body);
    const userId = req.user.idUser || req.user.sub;

    const resultado = await this.tandaService.unirseATanda(
      dto.inviteCode,
      userId,
      dto.selectedNumber,
    );

    return {
      statusCode: HttpStatus.OK,
      ...resultado,
    };
  }
}
